<script>
(()=>{
  const CFG=window.DPRO_CONFIG||{apiBaseUrl:"https://dpro-sales-line-api.dpromstk2000.workers.dev",sessionStorageKey:"dpro_sales_session_v2"};
  const $=s=>document.querySelector(s);
  let token="";
  let settings={};
  let staff=[];

  async function api(path,{method="GET",body=null}={}){
    const headers={Accept:"application/json"};
    if(body!==null)headers["Content-Type"]="application/json";
    if(token)headers.Authorization="Bearer "+token;
    const response=await fetch(CFG.apiBaseUrl+path,{method,headers,body:body===null?undefined:JSON.stringify(body),cache:"no-store"});
    const data=await response.json().catch(()=>({}));
    if(!response.ok||data.ok===false)throw new Error(data.message||data.error||"処理に失敗しました");
    return data;
  }
  function showError(message){$("#loginError").textContent=message;$("#loginError").classList.remove("hidden")}
  function toast(message){alert(message)}
  function esc(v){return String(v??"").replace(/[&<>'"]/g,c=>({"&":"&amp;","<":"&lt;",">":"&gt;","'":"&#39;",'"':"&quot;"}[c]))}
  async function login(e){
    e.preventDefault();
    $("#loginError").classList.add("hidden");
    try{
      const data=await api("/api/auth/login",{method:"POST",body:{facilityCode:$("#facilityCode").value.trim(),adminCode:$("#adminCode").value}});
      token=data.token;
      $("#loginPage").classList.add("hidden");
      $("#app").classList.remove("hidden");
      await loadAll();
    }catch(error){showError(error.message)}
  }
  async function loadAll(){
    const [settingsData,staffData]=await Promise.all([api("/api/team/settings"),api("/api/staff")]);
    settings=settingsData.settings||{};
    staff=staffData.staff||[];
    render();
  }
  function render(){
    $("#multiStaffEnabled").checked=settings.multiStaffEnabled===true;
    $("#claimTimeout").value=String(settings.claimTimeoutMinutes||180);
    $("#staffCanSeeAll").value=String(settings.staffCanSeeAllQueue!==false);
    $("#modeBadge").textContent=settings.multiStaffEnabled?"現在ON":"現在OFF";
    $("#modeBadge").className="badge "+(settings.multiStaffEnabled?"on":"off");
    $("#staffRows").innerHTML=staff.length?staff.map(s=>`<tr><td><strong>${esc(s.display_name)}</strong><small>${esc(s.staff_code)}</small></td><td>${esc(s.login_id||"未設定")}</td><td>${esc(s.role)}</td><td>${s.is_active?'<span class="badge on">有効</span>':'<span class="badge off">停止</span>'}</td><td><div class="actions"><button class="btn secondary small" data-pin="${s.id}" data-name="${esc(s.display_name)}">PIN変更</button>${s.role==="owner"?"":`<button class="btn ${s.is_active?"danger":"secondary"} small" data-active="${s.id}" data-next="${s.is_active?"false":"true"}">${s.is_active?"停止":"再開"}</button>`}</div></td></tr>`).join(""):'<tr><td colspan="5" class="empty">スタッフが登録されていません。</td></tr>';
  }
  async function saveSettings(){
    const enabled=$("#multiStaffEnabled").checked;
    if(!enabled&&settings.multiStaffEnabled&&!confirm("OFFにすると現在の担当確保が解除されます。よろしいですか？"))return;
    const data=await api("/api/team/settings",{method:"PATCH",body:{enabled,claimTimeoutMinutes:Number($("#claimTimeout").value),staffCanSeeAllQueue:$("#staffCanSeeAll").value==="true"}});
    settings=data.settings;
    render();
    toast(enabled?"複数スタッフ機能をONにしました":"1人運用へ戻しました");
  }
  async function createStaff(form){
    const data=Object.fromEntries(new FormData(form));
    await api("/api/staff",{method:"POST",body:{staffCode:data.staffCode,displayName:data.displayName,loginId:data.loginId,pin:data.pin,role:data.role,isActive:true}});
    form.reset();
    await loadAll();
    toast("スタッフを登録しました");
  }
  async function changePin(id,name){
    const pin=prompt(`${name}さんの新しい4桁PINを入力してください`);
    if(pin===null)return;
    if(!/^\d{4}$/.test(pin)){toast("PINは4桁の数字です");return}
    await api(`/api/staff/${id}`,{method:"PATCH",body:{pin}});
    toast("PINを変更しました");
  }
  async function changeActive(id,next){
    await api(`/api/staff/${id}`,{method:"PATCH",body:{isActive:next}});
    await loadAll();
  }
  $("#loginForm").addEventListener("submit",login);
  $("#saveSettings").onclick=()=>saveSettings().catch(e=>toast(e.message));
  $("#reloadStaff").onclick=()=>loadAll().catch(e=>toast(e.message));
  $("#staffForm").addEventListener("submit",e=>{e.preventDefault();createStaff(e.target).catch(err=>toast(err.message))});
  document.addEventListener("click",e=>{
    const pin=e.target.closest("[data-pin]");
    if(pin)changePin(pin.dataset.pin,pin.dataset.name).catch(err=>toast(err.message));
    const active=e.target.closest("[data-active]");
    if(active)changeActive(active.dataset.active,active.dataset.next==="true").catch(err=>toast(err.message));
  });
})();
</script>
